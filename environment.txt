# Conda environment setup for connected_thinking
# Use GPU variant if your machine has NVIDIA CUDA support.
#
# NOTE: export PYTHONNOUSERSITE=1 (or pass it inline) before running pip/python
# in this env if you have packages installed in ~/.local — otherwise your user
# site-packages can shadow the env's own packages.

# 1) Create and activate environment
conda create -n connected-thinking python=3.10 -y
conda activate connected-thinking

# 2A) GPU install (recommended when CUDA is available)
# For recent GPUs (e.g. Blackwell/RTX 50-series) conda's pytorch-cuda=12.1
# build is too old to support the hardware; install a CUDA 12.8 wheel via pip.
PYTHONNOUSERSITE=1 pip install --index-url https://download.pytorch.org/whl/cu128 torch
PYTHONNOUSERSITE=1 pip install numpy matplotlib pytest
